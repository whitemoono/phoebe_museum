```ts
// hooks/useArtworks.ts
export function useArtworks(params: ArtworkQuery) {
  const { data, error, isLoading, mutate } = useSWR(
    ['/artworks', params],
    ([url, p]) => apiClient.get(url, { params: p }),
    { revalidateOnFocus: false, dedupingInterval: 30000 }
  );
  return { artworks: data?.list, total: data?.total, isLoading, error, mutate };
}
```

## 5. API Client

```ts
// lib/api-client.ts
const apiClient = {
  async get<T>(url: string, opts?: { params?: Record<string, any> }): Promise<T> {
    const token = useAuthStore.getState().token;
    const res = await fetch(`/api${url}?${new URLSearchParams(opts?.params)}`, {
      headers: { Authorization: token ? `Bearer ${token}` : '' },
    });
    const json = await res.json();
    if (json.code !== 0) throw new ApiError(json.code, json.message);
    return json.data;
  },

  async post<T>(url: string, body?: any): Promise<T> {
    // 类似 get，method: 'POST'
  },

  async upload(file: File): Promise<string> {
    const { uploadUrl, cdnUrl } = await this.post('/upload/token', {
      filename: file.name, contentType: file.type, fileSize: file.size,
    });
    await fetch(uploadUrl, { method: 'PUT', body: file });
    return cdnUrl;
  },
};
```

## 8. 滚动渐入 Hook

```ts
// hooks/useReveal.ts
import { useEffect } from 'react';

export function useReveal() {
  useEffect(() => {
    const els = document.querySelectorAll('.rv');
    const io = new IntersectionObserver(
      (entries) => {
        entries.forEach((e) => {
          if (e.isIntersecting) {
            e.target.classList.add('in');
            io.unobserve(e.target);
          }
        });
      },
      { threshold: 0.06, rootMargin: '0px 0px -40px 0px' }
    );
    els.forEach((el) => io.observe(el));
    return () => io.disconnect();
  });
}
```

## 9. 性能优化

| 策略 | 实现 |
|------|------|
| 图片懒加载 | `loading="lazy"` + 缩略图 |
| 代码分割 | Next.js 自动 route-level splitting |
| SWR 缓存 | 列表数据 30s 去重，focus 不刷新 |
| ISR | 已发布作品静态生成 + 5min 增量 |
| 图片格式 | webp 优先，多尺寸 |
| font-display | `swap` 避免 FOIT |
| 骨架屏 | 列表加载时显示灰色占位 |
| CDN | 静态资源全 CDN 加速 |
