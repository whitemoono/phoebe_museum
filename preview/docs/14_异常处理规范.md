## 2. 后端错误处理

### 2.1 自定义错误类

```ts
// server/lib/errors.ts
export class ApiError extends Error {
  constructor(
    public code: number,
    public message: string,
    public httpStatus: number = 400,
    public details?: any,
  ) {
    super(message);
  }
}

// 便捷构造
export const Errors = {
  BadRequest: (msg: string, code = 40000) => new ApiError(code, msg, 400),
  Unauthorized: (msg = '请先登录', code = 40101) => new ApiError(code, msg, 401),
  Forbidden: (msg = '权限不足', code = 40301) => new ApiError(code, msg, 403),
  NotFound: (msg = '资源不存在', code = 40400) => new ApiError(code, msg, 404),
  Conflict: (msg: string, code = 40900) => new ApiError(code, msg, 409),
  RateLimited: (msg = '操作太频繁了') => new ApiError(42900, msg, 429),
  Internal: (msg = '服务器开小差了') => new ApiError(50001, msg, 500),
};
```

### 2.2 统一错误处理器

```ts
// server/middleware/error-handler.ts
export function handleError(err: unknown) {
  if (err instanceof ApiError) {
    return NextResponse.json(
      { code: err.code, message: err.message, data: null },
      { status: err.httpStatus },
    );
  }

  if (err instanceof ZodError) {
    return NextResponse.json(
      { code: 40001, message: err.errors[0]?.message || '参数校验失败', data: null },
      { status: 400 },
    );
  }

  // Prisma 错误映射
  if (err instanceof Prisma.PrismaClientKnownRequestError) {
    if (err.code === 'P2002') {
      const target = err.meta?.target as string[];
      return NextResponse.json(
        { code: 40905, message: `${target?.[0]} 已被占用`, data: null },
        { status: 409 },
      );
    }
    if (err.code === 'P2025') {
      return NextResponse.json(
        { code: 40401, message: '资源不存在', data: null },
        { status: 404 },
      );
    }
  }

  // 未知错误
  Sentry.captureException(err);
  return NextResponse.json(
    { code: 50001, message: '服务器开小差了', data: null },
    { status: 500 },
  );
}
```

## 3. 前端错误处理

### 3.1 API 请求拦截

```ts
// lib/api-client.ts
async function request<T>(url: string, options?: RequestInit): Promise<T> {
  const res = await fetch(url, options);
  const json = await res.json();

  if (json.code !== 0) {
    throw new ApiError(json.code, json.message);
  }
  return json.data;
}
```

### 3.2 全局错误处理

```ts
// hooks/useApi.ts
export function useApi<T>(fn: () => Promise<T>) {
  const [error, setError] = useState<ApiError | null>(null);

  const execute = useCallback(async () => {
    try {
      setError(null);
      return await fn();
    } catch (err) {
      if (err instanceof ApiError) {
        setError(err);
        // 全局 toast
        toast.error(err.message);
        // 401 → 跳转登录
        if (err.code >= 40100 && err.code < 40200) {
          redirectToLogin();
        }
      }
      throw err;
    }
  }, [fn]);

  return { execute, error };
}
```

## 4. Toast 通知

```ts
// 统一 UI 通知，替代 alert/confirm
const toast = {
  success: (msg: string) => showToast(msg, 'success'),
  error: (msg: string) => showToast(msg, 'error'),
  info: (msg: string) => showToast(msg, 'info'),
};

// 轻量实现
function showToast(message: string, type: 'success' | 'error' | 'info') {
  const el = document.createElement('div');
  el.className = `toast toast-${type}`;
  el.textContent = message;
  document.body.appendChild(el);
  requestAnimationFrame(() => el.classList.add('show'));
  setTimeout(() => {
    el.classList.remove('show');
    setTimeout(() => el.remove(), 300);
  }, 3000);
}
```

## 5. 日志记录

| 级别 | 场景 | 存储 |
|------|------|------|
| ERROR | 500 错误、数据库异常 | Sentry + 本地日志 |
| WARN | 403/429 | 本地日志 |
| INFO | 审核操作、用户注册 | 本地日志 |
| DEBUG | 开发环境全量 | 控制台 |

```ts
// server/lib/logger.ts
export const logger = {
  error: (msg: string, meta?: any) => { Sentry.captureMessage(msg); console.error(msg, meta); },
  warn: (msg: string, meta?: any) => console.warn(msg, meta),
  info: (msg: string, meta?: any) => console.info(msg, meta),
};
```
