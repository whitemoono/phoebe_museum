## 4. API Route 模式

```ts
// app/api/artworks/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { authMiddleware } from '@/server/middleware/auth';
import { rateLimit } from '@/server/middleware/rate-limit';
import { ArtworkService } from '@/server/services/artwork.service';
import { artworkQuerySchema } from '@/server/validators/artwork.validator';

export async function GET(req: NextRequest) {
  try {
    const params = artworkQuerySchema.parse(Object.fromEntries(req.nextUrl.searchParams));
    const result = await ArtworkService.list(params);
    return NextResponse.json({ code: 0, data: result });
  } catch (err) {
    return handleError(err);
  }
}

export async function POST(req: NextRequest) {
  try {
    const user = await authMiddleware(req);
    rateLimit(user.id, 'submit', 5, 3600);
    const formData = await req.formData();
    const artwork = await ArtworkService.submit(user.id, formData);
    return NextResponse.json({ code: 0, data: artwork }, { status: 201 });
  } catch (err) {
    return handleError(err);
  }
}
```

## 6. PM 编号分配

```ts
// server/lib/pm-counter.ts
export async function allocatePmNo(): Promise<string> {
  const result = await prisma.$queryRaw`
    UPDATE pm_counter
    SET next_pm_no = next_pm_no + 1, updated_at = NOW()
    WHERE id = 1
    RETURNING next_pm_no - 1 as allocated
  `;
  const next = (result[0] as any).allocated;
  return `PM-${String(next).padStart(6, '0')}`;
}
```

- 使用 PostgreSQL `FOR UPDATE` 行锁
- 审核通过时调用，事务内执行
- 保证编号唯一且递增

## 7. 认证方案

### 7.1 NextAuth.js 配置

```ts
// server/lib/auth.ts
import NextAuth from 'next-auth';

export const { auth, handlers } = NextAuth({
  providers: [
    CredentialsProvider({ ... }),     // 邮箱+密码
    GitHubProvider({ ... }),           // GitHub OAuth
    // WeChat OAuth 自定义 provider
  ],
  session: { strategy: 'jwt', maxAge: 7 * 24 * 60 * 60 },
  callbacks: {
    jwt: async ({ token, user }) => {
      if (user) { token.id = user.id; token.role = user.role; }
      return token;
    },
    session: async ({ session, token }) => {
      session.user.id = token.id as number;
      session.user.role = token.role as string;
      return session;
    },
  },
});
```

### 7.2 中间件

```ts
// server/middleware/auth.ts
export async function authMiddleware(req: NextRequest): Promise<User> {
  const session = await auth();
  if (!session?.user) {
    throw new ApiError(40101, '请先登录');
  }
  return session.user;
}

export async function adminMiddleware(req: NextRequest): Promise<User> {
  const user = await authMiddleware(req);
  if (user.role !== 'admin') {
    throw new ApiError(40301, '权限不足');
  }
  return user;
}
```

## 8. 限流方案

```ts
// server/middleware/rate-limit.ts
export async function rateLimit(key: string, action: string, max: number, windowSec: number) {
  const redisKey = `rate:${action}:${key}`;
  const count = await redis.incr(redisKey);
  if (count === 1) await redis.expire(redisKey, windowSec);
  if (count > max) {
    throw new ApiError(42900, '操作太频繁了，休息一下');
  }
}
```

## 9. 异步任务

### 9.1 缩略图生成

```ts
// 审核通过时入队
await thumbnailQueue.add('generate', { artworkId: 179 });

// Worker
new Worker('thumbnail', async (job) => {
  const { artworkId } = job.data;
  const media = await prisma.artworkMedia.findMany({ where: { artworkId } });
  for (const m of media) {
    const original = await fetchImage(m.url);
    const thumb = await sharp(original).resize(800).webp().toBuffer();
    const thumbUrl = await oss.upload(thumb, `thumb/${m.id}.webp`);
    await prisma.artworkMedia.update({ where: { id: m.id }, data: { thumbUrl } });
  }
});
```
