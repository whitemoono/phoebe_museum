```ts
// __tests__/services/artwork.service.test.ts
import { ArtworkService } from '@/server/services/artwork.service';
import { prisma } from '@/server/lib/prisma';

vi.mock('@/server/lib/prisma');

describe('ArtworkService', () => {
  describe('list()', () => {
    it('should return published artworks only', async () => {
      prisma.artwork.findMany.mockResolvedValue([]);
      prisma.artwork.count.mockResolvedValue(0);

      const result = await ArtworkService.list({ page: 1, pageSize: 20 });

      expect(prisma.artwork.findMany).toHaveBeenCalledWith(
        expect.objectContaining({ where: expect.objectContaining({ status: 'published' }) })
      );
      expect(result.total).toBe(0);
    });

    it('should filter by medium', async () => {
      await ArtworkService.list({ page: 1, pageSize: 20, medium: 'ai' });
      expect(prisma.artwork.findMany).toHaveBeenCalledWith(
        expect.objectContaining({ where: expect.objectContaining({ medium: 'ai' }) })
      );
    });
  });

  describe('submit()', () => {
    it('should create artwork with pending status', async () => {
      prisma.artwork.create.mockResolvedValue({ id: 1, status: 'pending' });
      const result = await ArtworkService.submit(231, mockFormData);
      expect(result.status).toBe('pending');
    });
  });
});
```

## 4. API 集成测试

```ts
// __tests__/api/artworks.test.ts
import { createTestClient } from './helpers';

describe('GET /api/artworks', () => {
  it('should return paginated published artworks', async () => {
    const res = await createTestClient().get('/api/artworks?page=1&pageSize=5');
    expect(res.status).toBe(200);
    expect(res.body.code).toBe(0);
    expect(res.body.data.list).toHaveLength(5);
    expect(res.body.data.list[0]).toHaveProperty('pmNo');
    expect(res.body.data.list[0].status).toBe('published');
  });

  it('should filter by medium=ai', async () => {
    const res = await createTestClient().get('/api/artworks?medium=ai');
    res.body.data.list.forEach(a => expect(a.medium).toBe('ai'));
  });
});

describe('POST /api/artworks/submit', () => {
  it('should require auth', async () => {
    const res = await createTestClient().post('/api/artworks/submit');
    expect(res.status).toBe(401);
    expect(res.body.code).toBe(40101);
  });

  it('should create pending artwork', async () => {
    const res = await createTestClient({ userId: 231 })
      .post('/api/artworks/submit')
      .field('title', '测试菲比')
      .field('medium', 'illustration')
      .attach('files', 'test.png');
    expect(res.status).toBe(201);
    expect(res.body.data.status).toBe('pending');
  });

  it('should reject without title', async () => {
    const res = await createTestClient({ userId: 231 })
      .post('/api/artworks/submit')
      .field('medium', 'illustration');
    expect(res.body.code).toBe(40001);
  });
});

describe('POST /api/artworks/:id/resonance', () => {
  it('should toggle resonance', async () => {
    const client = createTestClient({ userId: 231 });
    const res1 = await client.post('/api/artworks/1/resonance');
    expect(res1.body.data.hasResonanced).toBe(true);
    const res2 = await client.delete('/api/artworks/1/resonance');
    expect(res2.body.data.hasResonanced).toBe(false);
  });

  it('should reject duplicate', async () => {
    const client = createTestClient({ userId: 231 });
    await client.post('/api/artworks/1/resonance');
    const res = await client.post('/api/artworks/1/resonance');
    expect(res.body.code).toBe(40901);
  });
});
```

## 5. E2E 测试

```ts
// e2e/submit.spec.ts
import { test, expect } from '@playwright/test';

test.describe('投稿流程', () => {
  test('完整投稿到审核', async ({ page, browser }) => {
    // 1. 登录创作者
    await page.goto('/login');
    await page.fill('input[name=account]', 'test@phoebemuseum.art');
    await page.fill('input[name=password]', 'test1234');
    await page.click('button[type=submit]');
    await page.waitForURL('/my');

    // 2. 上交菲比
    await page.goto('/submit');
    await page.setInputFiles('input[type=file]', 'e2e/fixtures/test-phoebe.png');
    await page.fill('input[placeholder*="标题"]', 'E2E 测试菲比');
    await page.check('input[type=checkbox]');
    await page.click('button[type=submit]');
    await page.waitForURL('/my');

    // 3. 验证待审状态
    const status = await page.locator('.status.rev').first();
    await expect(status).toContainText('IN REVIEW');

    // 4. 审核员审核
    const adminPage = await browser.newPage();
    await adminPage.goto('/login');
    // ... 登录审核员
    await adminPage.goto('/admin/reviews');
    await adminPage.click('text=E2E 测试菲比');
    await adminPage.click('text=通过');

    // 5. 验证已发布
    await page.reload();
    const pubStatus = await page.locator('.status.pub').first();
    await expect(pubStatus).toContainText('PUBLISHED');
  });
});

test.describe('浏览流程', () => {
  test('从首页到作品详情', async ({ page }) => {
    await page.goto('/');
    await page.locator('.artwork-card').first().click();
    await page.waitForURL(/\/artwork\/PM-/);
    await expect(page.locator('.pm-no')).toBeVisible();
    await expect(page.locator('h1')).toBeVisible();
  });
});
```
